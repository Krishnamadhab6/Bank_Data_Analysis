A6	INT,
A7	TINYINT,
A8	TINYINT,
A9 	TINYINT,
A10	INT,
A11	INT,
A12	decimal(25,2),
A13	decimal(25,2),
A14	INT,
A15	BIGINT,
A16	BIGINT,
CONSTRAINT PK_Demographic PRIMARY KEY(district_id)

/*Note: currently birth number is loading as characters as it is enclosed by quotes*/

CREATE TABLE `client`
(
client_id INT,
`birth number` INT,
district_id INT,
CONSTRAINT PK_CLIENT PRIMARY KEY(CLIENT_ID),
CONSTRAINT FK_DEMOGRAPHIC_CLIENT FOREIGN KEY(DISTRICT_ID) REFERENCES `DEMOGRAPHIC`(district_id)
);


CREATE TABLE `ACCOUNT`
(
account_id INT,
district_id INT,
frequency VARCHAR(2000),
`DATE` DATE,
CONSTRAINT PK_ACCOUNT PRIMARY KEY(ACCOUNT_ID),
CONSTRAINT FK_ACCOUNT_DEMOGRAPHIC FOREIGN KEY(DISTRICT_ID) REFERENCES `DEMOGRAPHIC`(district_id)
);


CREATE TABLE `disp`
(
disp_id INT,
client_id INT,
account_id INT,
`TYPE` VARCHAR(50),
CONSTRAINT PK_DISPOSITION PRIMARY KEY(disp_id),
CONSTRAINT FK_DISPOSITION_CLIENT FOREIGN KEY(CLIENT_ID) REFERENCES `CLIENT`(CLIENT_ID),
CONSTRAINT FK_DISPOSITION_ACCOUNT FOREIGN KEY(ACCOUNT_ID) REFERENCES `ACCOUNT`(ACCOUNT_ID)
);

CREATE TABLE `Card`
(
card_id INT,
disp_id INT,
`TYPE` VARCHAR(50),
`issued` Date ,
CONSTRAINT PK_CreditCard PRIMARY KEY(CARD_ID),
CONSTRAINT FK_CREDITCARD_DISPOSITION FOREIGN KEY(DISP_ID) REFERENCES `disp`(DISP_ID)
);


CREATE TABLE `Loan`
(
loan_id INT,
account_id INT,
`DATE` DATE ,
amount DECIMAL(25,2),
duration INT,
payments DECIMAL(25,2),
`STATUS` VARCHAR(10),
CONSTRAINT PK_LOAN PRIMARY KEY(LOAN_ID) ,
CONSTRAINT FK_LOAN_ACCOUNT FOREIGN KEY(ACCOUNT_ID) REFERENCES `ACCOUNT`(account_ID)
);


CREATE TABLE `ordert`
(
order_id INT,
account_id INT,
bank_to VARCHAR(2),
account_to BIGINT,
amount DECIMAL(25,2),
K_symbol VARCHAR(20),
CONSTRAINT PK_order PRIMARY KEY(ORDER_ID),
CONSTRAINT FK_ORDER_ACCOUNT FOREIGN KEY(ACCOUNT_ID) REFERENCES `ACCOUNT`(ACCOUNT_ID)
);

CREATE TABLE `transaction`
(
trans_id INT,
account_id INT,
`DATE` DATE,
`type` VARCHAR(20),
operation VARCHAR(100),
amount DECIMAL(25,2),
balance DECIMAL(25,2),
k_symbol VARCHAR(50),
bank VARCHAR(50),
`account` BIGINT,
CONSTRAINT PK_TRANSACTION PRIMARY KEY(TRANS_ID) ,
CONSTRAINT FK_TRANSACTION_ACCOUNT FOREIGN KEY(aCCOUNT_ID) REFERENCES `ACCOUNT`(ACCOUNT_ID)
);
